# วันที่แจ้ง เลขที่ใบสั่งผลิต แผนก เครื่องจักร หัวหน้าเครื่อง ปัญหาที่พบ จำนวนที่พบ การจัดการ

เชื่อมต่อ Google Sheet (Apps Script)

ซิงค์ข้อมูลบันทึกและแก้ไขแบบเรียลไทม์อัตโนมัติ

นำ URL ที่ได้จากการ Deploy Web App (ลงท้ายด้วย /exec) มาวางที่นี่

เปิดใช้งาน Auto-Sync เรียลไทม์ ดึงข้อมูลอัตโนมัติจาก Sheet ทุกๆ 15 วินาที
สถานะปัจจุบัน: ทำงานในโหมด Local Storage (พร้อมใช้งานทันที) หากใส่ URL ด้านบน ระบบจะซิงค์ขึ้น Google Sheet ให้อัตโนมัติ

เครื่องมือจัดการข้อมูลทดสอบ

โค้ด Google Apps Script สำเร็จรูป (รองรับอ่าน / เพิ่ม / แก้ไข / ลบ)

วิธีติดตั้ง Google Apps Script ง่ายๆ ใน 4 ขั้นตอน:

  1. สร้างไฟล์ Google Sheets ใหม่ ตั้งชื่อตามต้องการ แล้วไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. ลบโค้ดเดิมทั้งหมด แล้ววางโค้ดด้านล่างนี้ลงไป จากนั้นกด บันทึก (Save)
  3. กดปุ่ม ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment) เลือกประเภทเป็น เว็บแอป (Web app)
  4. ตั้งค่าผู้มีสิทธิ์เข้าถึงเป็น "ทุกคน (Anyone)" กด Deploy แล้วคัดลอก Web app URL มาวางในช่องด้านซ้าย
// ============================================================================
// Google Apps Script สำหรับระบบติดตามปัญหาระหว่างการผลิต เหรียญทองการพิมพ์ (WG6)
// รองรับ: อ่านข้อมูล (GET), เพิ่มข้อมูลใหม่ (CREATE), แก้ไข (UPDATE), ลบ (DELETE)
// ============================================================================

function doGet(e) {
  var sheetName = (e && e.parameter && e.parameter.sheet) ? e.parameter.sheet : 'WG6_Defects';
  var sheet = getOrCreateSheet(sheetName);
  var data = sheet.getDataRange().getValues();
  var records = [];

  for (var i = 1; i < data.length; i++) {
    if (data[i][0] !== '') {
      records.push({
        id: String(data[i][0]),
        date: formatDateValue(data[i][1]),
        workOrder: String(data[i][2]),
        department: String(data[i][3]),
        machine: String(data[i][4]),
        operator: String(data[i][5]),
        defect: String(data[i][6]),
        quantity: Number(data[i][7]) || 0,
        unit: String(data[i][8] || 'แผ่น'),
        updatedAt: String(data[i][9] || '')
      });
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    records: records
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action || 'create';
    var sheetName = payload.sheetName || 'WG6_Defects';
    var sheet = getOrCreateSheet(sheetName);
    var rec = payload.record;

    if (action === 'create') {
      sheet.appendRow([
        rec.id, rec.date, rec.workOrder, rec.department,
        rec.machine, rec.operator, rec.defect, rec.quantity,
        rec.unit, new Date().toISOString()
      ]);
    } else if (action === 'update') {
      var rows = sheet.getDataRange().getValues();
      var found = false;
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(rec.id)) {
          sheet.getRange(i + 1, 1, 1, 10).setValues([[
            rec.id, rec.date, rec.workOrder, rec.department,
            rec.machine, rec.operator, rec.defect, rec.quantity,
            rec.unit, new Date().toISOString()
          ]]);
          found = true;
          break;
        }
      }
      if (!found) {
        sheet.appendRow([
          rec.id, rec.date, rec.workOrder, rec.department,
          rec.machine, rec.operator, rec.defect, rec.quantity,
          rec.unit, new Date().toISOString()
        ]);
      }
    } else if (action === 'delete') {
      var rowsDel = sheet.getDataRange().getValues();
      for (var j = rowsDel.length - 1; j >= 1; j--) {
        if (String(rowsDel[j][0]) === String(payload.id)) {
          sheet.deleteRow(j + 1);
          break;
        }
      }
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      message: 'บันทึกข้อมูลเรียบร้อย'
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(sheetName) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow([
      'ID', 'วันที่แจ้ง', 'เลขที่ใบสั่งผลิต', 'แผนก', 'เครื่องจักร',
      'หัวหน้าเครื่อง', 'ปัญหาที่พบ', 'จำนวนที่พบ', 'หน่วยนับ', 'อัปเดตล่าสุด'
    ]);
    sheet.getRange(1, 1, 1, 10).setFontWeight('bold').setBackground('#0F2544').setFontColor('#FBBF24');
  }
  return sheet;
}

function formatDateValue(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(val);
}
